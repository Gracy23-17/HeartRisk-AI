        // --- 2. Registry Filtering Logic ---
        function filterRegistry() {
            const ageVal = document.getElementById('searchAge').value;
            const bgVal = document.getElementById('searchBG').value;
            const out = document.getElementById('registryOutput');
            out.innerHTML = '';

            const results = PATIENTS.filter(p => 
                (ageVal === "" || p.age == ageVal) && 
                (bgVal === "" || p.bg === bgVal)
            );

            if (results.length === 0) {
                out.innerHTML = '<p style="color: #555;">No records found matching these criteria.</p>';
                return;
            }

            results.forEach(p => {
                const card = document.createElement('div');
                card.className = 'glass patient-card';
                card.innerHTML = `
                    <div class="tag">${p.id}</div>
                    <h3 style="color: var(--p);">${p.name}</h3>
                    <p style="font-size: 0.9rem; color: #888;">Age: ${p.age} | Blood Group: ${p.bg}</p>

                    <p style="text-align: center; font-weight: 700; color: var(--s); margin-top: 15px;">HVI Score: ${p.hvi}%</p>
                `;
                out.appendChild(card);
                renderRadar(p);
            });
        }

        // --- 3. Attractive Radar Chart for Registry ---
        function renderRadar(p) {
            const ctx = document.getElementById(`chart-${p.id}`).getContext('2d');
            new Chart(ctx, {
                type: 'radar',
                data: {
                    labels: ['Risk', 'BP', 'Chol', 'Stress'],
                    datasets: [{
                        label: 'Clinical Profile',
                        data: [p.hvi, (p.bp-100)/2, (p.chol-150)/2, p.stress],

                        borderColor: '#FF5E62',
                        pointBackgroundColor: '#FF5E62',
                        borderWidth: 2

                    plugins: { legend: { display: false } }
                }
            });
        }

        // --- 4. Predictive Live Chart ---
        function calculateLiveRisk() {
            const age = parseInt(document.getElementById('age').value);
            const bp = parseInt(document.getElementById('bp').value);
            const chol = parseInt(document.getElementById('chol').value);

            let risk = 5;
            if (age > 50) risk += (age-50)*0.8;
            if (bp > 130) risk += (bp-130)*0.4;
            if (chol > 200) risk += (chol-200)*0.2;
            
            const score = Math.min(Math.max(risk, 5), 95).toFixed(1);
            document.getElementById('riskTitle').innerText = score + "%";
            document.getElementById('riskDesc').innerText = score > 40 ? "🚨 High Priority" : "✅ Stable";

            const ctx = document.getElementById('liveChart').getContext('2d');
            if (charts.live) charts.live.destroy();
            charts.live = new Chart(ctx, {
                type: 'polarArea',
                data: {
                    labels: ['Overall Risk', 'Age Factor', 'BP Stress', 'Cholesterol'],
                    datasets: [{
                        data: [score, Math.min(age, 100), Math.min(bp-100, 100), Math.min(chol-150, 100)],
                        backgroundColor: ['rgba(255, 94, 98, 0.5)', 'rgba(255, 153, 102, 0.3)', 'rgba(255, 255, 255, 0.05)', 'rgba(255, 255, 255, 0.1)']
                    }]
                },
                options: {
                    responsive: true, maintainAspectRatio: false,
                    scales: { r: { grid: { color: 'rgba(255,255,255,0.05)' }, ticks: { display: false } } },
                    plugins: { legend: { labels: { color: '#aaa', font: { size: 12 } } } }
                }
            });
        }

            filterRegistry();

# HeartRisk AI: Cardiovascular Predictive Insights Platform

## Project Overview
**HeartRisk AI** is an advanced healthcare analytics solution designed to provide real-time, data-driven cardiovascular risk assessments. The project addresses the critical need for accessible preliminary health screening tools by integrating clinical data with a rule-based predictive model. It serves as a comprehensive portfolio project for medical technology internships, showcasing a full-stack development approach within a singular, high-performance architecture.

## Core Technical Features
- **Predictive Risk Model**: Implements a customized Cardiovascular Risk Index (CRI), a multi-factor algorithm that weighs key health indicators (Age, Systolic Blood Pressure, and Cholesterol) to estimate a 10-year probability of cardiovascular complications.
- **Dynamic Risk Profiling**: Utilizing a real-time reactive interface, the platform provides instantaneous risk categorization (Low, Moderate, High) and visual feedback through interactive data visualizations.
- **Zero-Latency Architecture**: Developed as a self-contained single-file application (`index.html`), the platform ensures rapid deployment and high reliability across all modern web environments with zero external backend dependencies.
- **Advanced UI/UX Design**: Featuring a premium frosted-glass design language, the platform offers a professional-grade user experience that matches modern medical technology standards.

## Objective & Impact
The primary objective of HeartRisk AI is to demonstrate the practical application of AI-driven logic in a healthcare context. By transforming raw clinical vitals into actionable health intelligence, the project highlights the developer's proficiency in algorithm design, data engineering, and high-fidelity front-end development—making it a flagship addition to a professional GitHub portfolio and medical internship application.
